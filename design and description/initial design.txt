@startuml VoleMachineSimulator
skinparam classAttributeIconSize 0
skinparam linetype ortho
title Vole Machine Simulator - Initial Class Design

class UI {
  - machine : VoleMachine
  --
  + showMenu() : void
  + takeInput() : void
}

class VoleMachine {
  - cpu : CPU
  - mem : Memory
  - isOneByOne : bool
  --
  + setIsOneByOne(mode : bool) : void
  + LoadFile(fileName : string) : bool
  + Execute() : void
  + output() : void
}

class CPU {
  - reg : Register
  - alu : ALU
  - pc : int
  - IR : string
  --
  + CPU()
  + Fetch(mem : Memory&) : void
  + Execute(mem : Memory&) : void
}

class Memory {
  - memory : vector<string>
  --
  + Memory()
  + get(address : int) : string
  + set(address : int, value : string) : void
}

class Register {
  - registers : vector<string>
  --
  + Register()
  + get(index : int) : string
  + set(index : int, value : string) : void
}

class ALU {
  - opCodes : vector<string>
  --
  + ALU()
  + hexToDec(hex : string) : int
  + decToHex(dec : int) : string
  + validOP(op : string) : bool
  + hexToFloat(hex : string) : float
  + floatToHex(value : float) : string
}

' ---------- Relationships ----------
UI "1" --> "1" VoleMachine : controls
VoleMachine "1" *-- "1" CPU : contains
VoleMachine "1" *-- "1" Memory : contains
CPU "1" *-- "1" Register : owns
CPU "1" *-- "1" ALU : owns
CPU ..> Memory : fetches / reads / writes

' ---------- Notes ----------
note right of Memory
  256 cells ("00".."FF"), 8 bits each.
  Address FF is the screen output address.
end note

note right of Register
  16 registers (R0..RF),
  each 8 bits (2 hex digits).
end note

note right of CPU
  IR is 4 hex digits ("0000" initially).
  Fetch advances PC by 2.
  Execute handles jumps and halt.
end note

note bottom of ALU
  Supports opcodes 1 through C.
end note

note bottom of VoleMachine
  Run mode: execute until halt.
  Step mode: one instruction per Enter.
  Programs load at even addresses.
end note
@enduml
